cmake_minimum_required(VERSION 3.21)
project(Tideglass LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Water requires a 32-bit Windows target")
endif()

set(STORM_REVISION 116135c27e8c0dcf8ea0a0c549a6d737ce10fc73)
set(STORM_SOURCE "" CACHE PATH "Optional local checkout of the pinned StormLib revision")
set(STORM_USE_BUNDLED_LIBRARIES ON CACHE BOOL "" FORCE)
set(STORM_UNICODE ON CACHE BOOL "" FORCE)
set(STORM_SKIP_INSTALL ON CACHE BOOL "" FORCE)
set(STORM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(STORM_SOURCE)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${STORM_SOURCE}"
                  OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${STORM_SOURCE}"
                  OUTPUT_VARIABLE changes COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL STORM_REVISION OR changes)
    message(FATAL_ERROR "StormLib must be clean at ${STORM_REVISION}")
  endif()
  add_subdirectory("${STORM_SOURCE}" stormlib EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(stormlib GIT_REPOSITORY https://github.com/ladislav-zezula/StormLib.git
                                GIT_TAG ${STORM_REVISION})
  FetchContent_MakeAvailable(stormlib)
endif()

get_target_property(storm_source_dir storm SOURCE_DIR)
target_compile_options(storm PRIVATE "-ffile-prefix-map=${storm_source_dir}=StormLib")
# The core's sources sit in folders under src/; the export never includes the standalone proxy.
file(GLOB_RECURSE units CONFIGURE_DEPENDS src/*.cpp)
add_library(water-core SHARED ${units})
set_target_properties(water-core PROPERTIES PREFIX "" CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_include_directories(water-core PRIVATE src generated "$<TARGET_PROPERTY:storm,SOURCE_DIR>/src")
target_compile_definitions(water-core PRIVATE WATER_COOPERATIVE_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(water-core PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=tideglass" -O2 -msse2 -mfpmath=sse -Wall -Wextra -Wno-unused-parameter -Werror)
target_link_options(water-core PRIVATE -static -s -Wl,--no-insert-timestamp)
target_link_libraries(water-core PRIVATE "$<LINK_ONLY:storm>" wininet advapi32 dxguid uuid)

# A changed decoder, dependency or compiler must never reuse another build's decoded textures.
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/source.json" source_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" recipe_hash)
string(SHA256 cache_recipe "${source_hash};${recipe_hash};${STORM_REVISION};${CMAKE_CXX_COMPILER_VERSION}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source.json")
target_compile_definitions(water-core PRIVATE WATER_TEXTURE_RECIPE="${cache_recipe}")
